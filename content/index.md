(function() {
  function initDependencySimulation() {
    const plate = document.querySelector('.dependency-plate');
    if (!plate || plate.dataset.depReady === 'true') return;
    plate.dataset.depReady = 'true';

    const nodes = Array.from(plate.querySelectorAll('.dep-node'));
    const lines = Array.from(plate.querySelectorAll('.dep-line'));
    const note = plate.querySelector('.failure-note');
    if (!nodes.length || !note) return;

    const strong = note.querySelector('strong');
    const small = note.querySelector('small');

    const scenarios = {
      '4': {
        title: 'Om infrastrukturen faller bort:',
        desc: 'El eller nätverk dör → systemen blir oåtkomliga → medarbetare tvingas improvisera → verksamhetens arbete stannar.'
      },
      '3': {
        title: 'Om systemet slås ut:',
        desc: 'Verktyget blir otillgängligt → handläggning måste skötas analogt → verksamhetens kapacitet faller drastiskt.'
      },
      '2': {
        title: 'Om människorna drabbas:',
        desc: 'Vid kris, hög stress eller otydligt mandat hjälper inte tekniken – samordning och beslutsfattande avgör utfallet.'
      },
      '1': {
        title: 'Verksamhetens kärnuppgift:',
        desc: 'Börja alltid här: identifiera vad som absolut inte får stanna, och härled de tekniska och mänskliga behoven bakåt.'
      }
    };

    let pinned = null;

    function applyState(nodeVal) {
      if (!nodeVal) {
        nodes.forEach(btn => {
          btn.classList.remove('is-active', 'is-impacted');
          btn.removeAttribute('aria-current');
        });
        lines.forEach(l => l.classList.remove('is-active'));
        if (strong) strong.textContent = 'Vad händer om X slutar fungera?';
        if (small) small.textContent = 'Klicka eller peka på en nivå för att följa beroendet tillbaka till verksamheten.';
        return;
      }

      const num = parseInt(nodeVal, 10);
      nodes.forEach(btn => {
        const n = parseInt(btn.dataset.node, 10);
        btn.classList.remove('is-active', 'is-impacted');
        if (n === num) {
          btn.classList.add('is-active');
          btn.setAttribute('aria-current', 'true');
        } else if (n < num) {
          btn.classList.add('is-impacted');
          btn.removeAttribute('aria-current');
        } else {
          btn.removeAttribute('aria-current');
        }
      });

      lines.forEach((line, idx) => {
        // idx 0 connects 1 & 2; idx 1 connects 2 & 3; idx 2 connects 3 & 4
        if (num >= 2 + idx) {
          line.classList.add('is-active');
        } else {
          line.classList.remove('is-active');
        }
      });

      const sc = scenarios[nodeVal];
      if (sc) {
        if (strong) strong.textContent = sc.title;
        if (small) small.textContent = sc.desc;
      }
    }

    nodes.forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.preventDefault();
        pinned = (pinned === btn.dataset.node) ? null : btn.dataset.node;
        applyState(pinned);
      });

      btn.addEventListener('mouseenter', () => {
        if (!pinned) applyState(btn.dataset.node);
      });

      btn.addEventListener('mouseleave', () => {
        if (!pinned) applyState(null);
      });
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initDependencySimulation);
  } else {
    initDependencySimulation();
  }
  document.addEventListener('nav', initDependencySimulation);
})();
</script>
